    73	static int bcm54xx_config_clock_delay(struct phy_device *phydev)
    74	{
    75		int rc, val;
    76	
    77		/* handling PHY's internal RX clock delay */
    78		val = bcm54xx_auxctl_read(phydev, MII_BCM54XX_AUXCTL_SHDWSEL_MISC);
    79		val |= MII_BCM54XX_AUXCTL_MISC_WREN;
    80		if (phydev->interface == PHY_INTERFACE_MODE_RGMII ||
    81		    phydev->interface == PHY_INTERFACE_MODE_RGMII_TXID) {
    82			/* Disable RGMII RXC-RXD skew */
    83			val &= ~MII_BCM54XX_AUXCTL_SHDWSEL_MISC_RGMII_SKEW_EN;
    84		}
    85		if (phydev->interface == PHY_INTERFACE_MODE_RGMII_ID ||
    86		    phydev->interface == PHY_INTERFACE_MODE_RGMII_RXID) {
    87			/* Enable RGMII RXC-RXD skew */
    88			val |= MII_BCM54XX_AUXCTL_SHDWSEL_MISC_RGMII_SKEW_EN;
    89		}
    90		rc = bcm54xx_auxctl_write(phydev, MII_BCM54XX_AUXCTL_SHDWSEL_MISC,
    91					  val);
    92		if (rc < 0)
    93			return rc;
    94	
    95		/* handling PHY's internal TX clock delay */
    96		val = bcm_phy_read_shadow(phydev, BCM54810_SHD_CLK_CTL);
    97		if (phydev->interface == PHY_INTERFACE_MODE_RGMII ||
    98		    phydev->interface == PHY_INTERFACE_MODE_RGMII_RXID) {
    99			/* Disable internal TX clock delay */
   100			val &= ~BCM54810_SHD_CLK_CTL_GTXCLK_EN;
   101		}
   102		if (phydev->interface == PHY_INTERFACE_MODE_RGMII_ID ||
   103		    phydev->interface == PHY_INTERFACE_MODE_RGMII_TXID) {
   104			/* Enable internal TX clock delay */
   105			val |= BCM54810_SHD_CLK_CTL_GTXCLK_EN;
   106		}
   107		rc = bcm_phy_write_shadow(phydev, BCM54810_SHD_CLK_CTL, val);
   108		if (rc < 0)
   109			return rc;
   110	
   111		return 0;
   112	}

   114	static int bcm54210e_config_init(struct phy_device *phydev)
   115	{
   116		int val;
   117	
   118		bcm54xx_config_clock_delay(phydev);
   119	
   120		if (phydev->dev_flags & PHY_BRCM_EN_MASTER_MODE) {
   121			val = phy_read(phydev, MII_CTRL1000);
   122			val |= CTL1000_AS_MASTER | CTL1000_ENABLE_MASTER;

   436	static int bcm54xx_config_init(struct phy_device *phydev)
   437	{
   438		int reg, err, val;
   439		u32 led_modes[] = {BCM_LED_MULTICOLOR_LINK_ACT,
   440				   BCM_LED_MULTICOLOR_LINK};
   441		struct device_node *np = phydev->mdio.dev.of_node;
   442	
   443		reg = phy_read(phydev, MII_BCM54XX_ECR);
   444		if (reg < 0)
   445			return reg;
   446	
   447		/* Initially all interrupts are masked in IMR, so unmask events
   448		 * we are interested in.
   449		 */
   450		reg = ~(MII_BCM54XX_INT_DUPLEX |
   451			MII_BCM54XX_INT_SPEED |
   452			MII_BCM54XX_INT_LINK);
   453		err = phy_write(phydev, MII_BCM54XX_IMR, reg);
   454		if (err < 0)
   455			return err;
   456	
   457		if ((BRCM_PHY_MODEL(phydev) == PHY_ID_BCM50610 ||
   458		     BRCM_PHY_MODEL(phydev) == PHY_ID_BCM50610M) &&
   459		    (phydev->dev_flags & PHY_BRCM_CLEAR_RGMII_MODE))
   460			bcm_phy_write_shadow(phydev, BCM54XX_SHD_RGMII_MODE, 0);
   461	
   462		if (of_property_read_bool(np, "brcm,powerdown-enable"))
   463			phydev->dev_flags |= PHY_BRCM_AUTO_PWRDWN_ENABLE;
   464	
   465		bcm54xx_adjust_rxrefclk(phydev);
   466	
   467		switch (BRCM_PHY_MODEL(phydev)) {
   468		case PHY_ID_BCM50610:
   469		case PHY_ID_BCM50610M:
   470			err = bcm54xx_config_clock_delay(phydev);
   471			break;
   472		case PHY_ID_BCM54210E:
   473			err = bcm54210e_config_init(phydev);
   474			break;
   475		case PHY_ID_BCM54612E:
   476			err = bcm54612e_config_init(phydev);
   477			break;
   478		case PHY_ID_BCM54213PE:
   479			err = bcm54213pe_config_init(phydev);
   480			break;
   481		case PHY_ID_BCM54616S:
   482			err = bcm54616s_config_init(phydev);
   483			break;
   484		case PHY_ID_BCM54810:
   485			/* For BCM54810, we need to disable BroadR-Reach function */
   486			val = bcm_phy_read_exp(phydev,
   487					       BCM54810_EXP_BROADREACH_LRE_MISC_CTL);
   488			val &= ~BCM54810_EXP_BROADREACH_LRE_MISC_CTL_EN;
   489			err = bcm_phy_write_exp(phydev,
   490						BCM54810_EXP_BROADREACH_LRE_MISC_CTL,
   491						val);
   492			break;
   493		case PHY_ID_BCM54811:
   494			err = bcm54811_config_init(phydev);
   495			break;
   496		}
   497		if (err)
   498			return err;
   499	
   500		bcm54xx_phydsp_config(phydev);
   501	
   502		of_property_read_u32_array(np, "led-modes", led_modes, 2);
   503	
   504		/* For non-SFP setups, encode link speed into LED1 and LED3 pair
   505		 * (green/amber).
   506		 * Don't do this for devices on an SFP module, since some of these
   507		 * use the LED outputs to control the SFP LOS signal, and changing
   508		 * these settings will cause LOS to malfunction.
   509		 */
   510		if (!phy_on_sfp(phydev)) {
   511			int led_swap = of_property_read_bool(np, "led-swap") ? 1 : 0;
   512			val = BCM54XX_SHD_LEDS1_LED1(BCM_LED_SRC_MULTICOLOR1) |
   513				BCM54XX_SHD_LEDS1_LED3(BCM_LED_SRC_MULTICOLOR1);
   514			bcm_phy_write_shadow(phydev, BCM54XX_SHD_LEDS1, val);
   515			/* BCM54210PE controls two extra LEDs with the next register.
   516			 * Make LED3 shadow LED1, but preserve LED4 as is - useful on
   517			 * CM4/CM5 which use LED3 for ETH_LEDY instead of LED1. LED4
   518			 * is either unused or configured as INT pin on CM5.
   519			 */
   520			reg = bcm_phy_read_shadow(phydev, BCM54XX_SHD_LEDS2);
   521			reg &= ~(0xf << 0);
   522			reg |= BCM54XX_SHD_LEDS1_LED1(BCM_LED_SRC_MULTICOLOR1);
   523			bcm_phy_write_shadow(phydev, BCM54XX_SHD_LEDS2, reg);
   524	
   525			val = BCM_LED_MULTICOLOR_IN_PHASE |
   526				BCM54XX_SHD_LEDS1_LED1(led_modes[0 ^ led_swap]) |
   527				BCM54XX_SHD_LEDS1_LED3(led_modes[1 ^ led_swap]);
   528			bcm_phy_write_exp(phydev, BCM_EXP_MULTICOLOR, val);
   529		}
   530	
   531		bcm54xx_ptp_config_init(phydev);
   532	
   533		/* Acknowledge any left over interrupt and charge the device for
   534		 * wake-up.
   535		 */
   536		err = bcm_phy_read_exp(phydev, BCM54XX_WOL_INT_STATUS);
   537		if (err < 0)
   538			return err;
   539	
   540		if (err)
   541			pm_wakeup_event(&phydev->mdio.dev, 0);
   542	
   543		return 0;
   544	}

   546	static int bcm54xx_iddq_set(struct phy_device *phydev, bool enable)
   547	{
   548		int ret = 0;
   549	
   550		if (!(phydev->dev_flags & PHY_BRCM_IDDQ_SUSPEND))
   551			return ret;
   552	
   553		ret = bcm_phy_read_exp(phydev, BCM54XX_TOP_MISC_IDDQ_CTRL);
   554		if (ret < 0)
   555			goto out;
   556	
   557		if (enable)
   558			ret |= BCM54XX_TOP_MISC_IDDQ_SR | BCM54XX_TOP_MISC_IDDQ_LP;
   559		else
   560			ret &= ~(BCM54XX_TOP_MISC_IDDQ_SR | BCM54XX_TOP_MISC_IDDQ_LP);
   561	
   562		ret = bcm_phy_write_exp(phydev, BCM54XX_TOP_MISC_IDDQ_CTRL, ret);
   563	out:
   564		return ret;
   565	}

   567	static int bcm54xx_set_wakeup_irq(struct phy_device *phydev, bool state)
   568	{
   569		struct bcm54xx_phy_priv *priv = phydev->priv;
   570		int ret = 0;
   571	
   572		if (!bcm54xx_phy_can_wakeup(phydev))
   573			return ret;
   574	
   575		if (priv->wake_irq_enabled != state) {
   576			if (state)
   577				ret = enable_irq_wake(priv->wake_irq);
   578			else
   579				ret = disable_irq_wake(priv->wake_irq);
   580			priv->wake_irq_enabled = state;
   581		}
   582	
   583		return ret;
   586	static int bcm54xx_suspend(struct phy_device *phydev)
   587	{
   588		int ret = 0;
   589	
   590		bcm54xx_ptp_stop(phydev);
   591	
   592		/* Acknowledge any Wake-on-LAN interrupt prior to suspend */
   593		ret = bcm_phy_read_exp(phydev, BCM54XX_WOL_INT_STATUS);
   594		if (ret < 0)
   595			return ret;
   596	
   597		if (phydev->wol_enabled)
   598			return bcm54xx_set_wakeup_irq(phydev, true);
   599	
   600		/* We cannot use a read/modify/write here otherwise the PHY gets into
   601		 * a bad state where its LEDs keep flashing, thus defeating the purpose
   602		 * of low power mode.
   603		 */
   604		ret = phy_write(phydev, MII_BMCR, BMCR_PDOWN);
   605		if (ret < 0)
   606			return ret;
   607	
   608		return bcm54xx_iddq_set(phydev, true);
   609	}

   611	static int bcm54xx_resume(struct phy_device *phydev)
   612	{
   613		int ret = 0;
   614	
   615		if (phydev->wol_enabled) {
   616			ret = bcm54xx_set_wakeup_irq(phydev, false);
   617			if (ret)
   618				return ret;
   619		}
   620	
   621		ret = bcm54xx_iddq_set(phydev, false);
   622		if (ret < 0)
   623			return ret;
   624	
   625		/* Writes to register other than BMCR would be ignored
   626		 * unless we clear the PDOWN bit first
   627		 */
   628		ret = genphy_resume(phydev);
   629		if (ret < 0)
   630			return ret;
   631	
   632		/* Upon exiting power down, the PHY remains in an internal reset state
   633		 * for 40us
   634		 */
   635		fsleep(40);
   636	
   637		/* Issue a soft reset after clearing the power down bit
   638		 * and before doing any other configuration.
   639		 */
   640		if (phydev->dev_flags & PHY_BRCM_IDDQ_SUSPEND) {
   641			ret = genphy_soft_reset(phydev);
   642			if (ret < 0)
   643				return ret;
   644		}
   645	
   646		return bcm54xx_config_init(phydev);
   647	}
