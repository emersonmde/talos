  2343	int genphy_read_master_slave(struct phy_device *phydev)
  2344	{
  2345		int cfg, state;
  2346		int val;
  2347	
  2348		phydev->master_slave_get = MASTER_SLAVE_CFG_UNKNOWN;
  2349		phydev->master_slave_state = MASTER_SLAVE_STATE_UNKNOWN;
  2350	
  2351		val = phy_read(phydev, MII_CTRL1000);
  2352		if (val < 0)
  2353			return val;
  2354	
  2355		if (val & CTL1000_ENABLE_MASTER) {
  2356			if (val & CTL1000_AS_MASTER)
  2357				cfg = MASTER_SLAVE_CFG_MASTER_FORCE;
  2358			else
  2359				cfg = MASTER_SLAVE_CFG_SLAVE_FORCE;
  2360		} else {
  2361			if (val & CTL1000_PREFER_MASTER)
  2362				cfg = MASTER_SLAVE_CFG_MASTER_PREFERRED;
  2363			else
  2364				cfg = MASTER_SLAVE_CFG_SLAVE_PREFERRED;
  2365		}
  2366	
  2367		val = phy_read(phydev, MII_STAT1000);
  2368		if (val < 0)
  2369			return val;
  2370	
  2371		if (val & LPA_1000MSFAIL) {
  2372			state = MASTER_SLAVE_STATE_ERR;
  2373		} else if (phydev->link) {
  2374			/* this bits are valid only for active link */
  2375			if (val & LPA_1000MSRES)
  2376				state = MASTER_SLAVE_STATE_MASTER;
  2377			else
  2378				state = MASTER_SLAVE_STATE_SLAVE;
  2379		} else {
  2380			state = MASTER_SLAVE_STATE_UNKNOWN;
  2381		}
  2382	
  2383		phydev->master_slave_get = cfg;
  2384		phydev->master_slave_state = state;
  2385	
  2386		return 0;
  2387	}
