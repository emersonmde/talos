    42	int __bcm_phy_read_exp(struct phy_device *phydev, u16 reg)
    43	{
    44		int val;
    45	
    46		val = __phy_write(phydev, MII_BCM54XX_EXP_SEL, reg);
    47		if (val < 0)
    48			return val;
    49	
    50		val = __phy_read(phydev, MII_BCM54XX_EXP_DATA);
    51	
    52		/* Restore default value.  It's O.K. if this write fails. */
    53		__phy_write(phydev, MII_BCM54XX_EXP_SEL, 0);
    54	
    55		return val;
    56	}
    93	int bcm54xx_auxctl_read(struct phy_device *phydev, u16 regnum)
